cmake_minimum_required(VERSION 3.20)
project(LowMemPiPNN VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|amd64|AMD64")
  message(FATAL_ERROR "This release targets Linux x86-64 with AVX-512 F/BW/DQ/VL and F16C")
endif()
find_package(OpenMP REQUIRED)
find_package(Threads REQUIRED)
find_package(Eigen3 3.4 REQUIRED NO_MODULE)
include(FetchContent)
set(ABSL_PROPAGATE_CXX_STD ON CACHE BOOL "" FORCE)
set(ABSL_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(HWY_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(HWY_ENABLE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(HWY_ENABLE_CONTRIB ON CACHE BOOL "" FORCE)
FetchContent_Declare(abseil_cpp GIT_REPOSITORY https://github.com/abseil/abseil-cpp.git GIT_TAG bc257a88f7c1939f24e0379f14a3589e926c950c UPDATE_DISCONNECTED TRUE)
FetchContent_Declare(highway GIT_REPOSITORY https://github.com/google/highway.git GIT_TAG 9d1374260ad6f234aa7bafcfccf48abe96c41999 UPDATE_DISCONNECTED TRUE)
FetchContent_MakeAvailable(abseil_cpp highway)
set(LOWMEM_ISA -march=native -mavx512f -mavx512bw -mavx512dq -mavx512vl -mavx2 -mfma -mf16c)
add_library(lowmem_options INTERFACE)
target_compile_options(lowmem_options INTERFACE -O3 ${LOWMEM_ISA})
target_link_libraries(lowmem_options INTERFACE OpenMP::OpenMP_CXX Threads::Threads Eigen3::Eigen)
target_include_directories(lowmem_options INTERFACE include src third_party/rabitq/include third_party/pipnn/include third_party/pipnn/include/bench third_party/parlay/include)
file(GLOB RABITQ_SIMD CONFIGURE_DEPENDS third_party/rabitq/src/simd/*.cpp)
add_library(lowmem_quant STATIC ${RABITQ_SIMD} third_party/rabitq/src/utils/cpu_features.cpp)
target_link_libraries(lowmem_quant PUBLIC lowmem_options)
add_library(lowmem_pre STATIC src/lowmem/pre_rabitq.cpp)
target_link_libraries(lowmem_pre PUBLIC lowmem_quant)
add_library(lowmem_leaf STATIC src/lowmem/ordered_leaf_stream.cpp)
target_compile_options(lowmem_leaf PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(lowmem_leaf PUBLIC lowmem_quant)
foreach(kind IN ITEMS float int8)
  add_library(lowmem_capture_${kind} STATIC src/lowmem/capture.cpp)
  target_compile_definitions(lowmem_capture_${kind} PRIVATE HWY_STATIC_DEFINE PARLAY_ELASTIC_PARALLELISM=true EIGEN_DONT_PARALLELIZE=1 TOOLCHAIN_MISS_ASM_HWCAP_H)
  target_compile_options(lowmem_capture_${kind} PRIVATE -ffast-math)
  target_link_libraries(lowmem_capture_${kind} PUBLIC lowmem_options absl::synchronization absl::flat_hash_set absl::flat_hash_map hwy hwy_contrib)
  add_executable(lowmem-pipnn-${kind} src/main.cpp)
  target_compile_options(lowmem-pipnn-${kind} PRIVATE -fno-fast-math)
  target_compile_definitions(lowmem-pipnn-${kind} PRIVATE EIGEN_DONT_PARALLELIZE=1)
  target_link_libraries(lowmem-pipnn-${kind} PRIVATE lowmem_capture_${kind} lowmem_pre lowmem_leaf rt)
  if(kind STREQUAL "int8")
    target_compile_definitions(lowmem_capture_${kind} PRIVATE LOWMEM_INT8_L2=1 B_UNIQUE_SIGNED_CARRIER=1)
    target_compile_definitions(lowmem-pipnn-${kind} PRIVATE LOWMEM_INT8_L2=1)
  endif()
endforeach()
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME smoke COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/smoke.py --build-dir ${CMAKE_BINARY_DIR})
